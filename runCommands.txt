====================================================================
             WEBDRIVERIO + CUCUMBER - RUN COMMANDS

Repository     : https://github.com/VijayanGit2026/WebdriverIO_Cucumber.git
Branch         : main
WebdriverIO    : 10.2.0
Shell          : Windows PowerShell

--------------------------------------------------------------------
1. FIRST-TIME CLONE AND SETUP
--------------------------------------------------------------------

# Navigate to the parent folder
cd "D:\Automation\Playwright"

# Clone the repository
git clone https://github.com/VijayanGit2026/WebdriverIO_Cucumber.git

# Navigate to the project folder
cd "D:\Automation\Playwright\WebdriverIO_Cucumber"

# Switch to main and get the latest code
git switch main
git pull origin main

# Check Node.js and npm versions
node -v
npm -v

# Install project dependencies
npm ci

# Verify WebdriverIO version and dependency security
npx wdio --version
npm audit


--------------------------------------------------------------------
2. RUN TESTS
--------------------------------------------------------------------

# Run all 6 test scenarios
npm test

# Run login scenarios:
# Valid login, locked-out account, invalid password, blank username
npm run test:login

# Run checkout scenarios:
# Complete checkout and remove backpack from cart
npm run test:checkout

# Run smoke scenarios:
# Valid login and complete checkout
npm run test:smoke

# Run scenarios tagged @regression
npm run test:regression


--------------------------------------------------------------------
3. RUN TESTS BY INDIVIDUAL CUCUMBER TAG
--------------------------------------------------------------------

# Positive login
npx wdio run wdio.conf.js --cucumberOpts.tags="@positive"

# Negative login scenarios
npx wdio run wdio.conf.js --cucumberOpts.tags="@negative"

# End-to-end checkout
npx wdio run wdio.conf.js --cucumberOpts.tags="@e2e"

# Regression scenarios
npx wdio run wdio.conf.js --cucumberOpts.tags="@regression"


--------------------------------------------------------------------
4. VIEW DOWNLOADED PDF FILES
--------------------------------------------------------------------

# List generated order PDFs
Get-ChildItem ".\pdf_download" -Filter "*.pdf"


--------------------------------------------------------------------
5. GENERATE AND OPEN ALLURE REPORT
--------------------------------------------------------------------

# Generate the Allure HTML report
npm run report:allure

# Open the Allure report
npm run report:open
